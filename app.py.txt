def filter_and_sort_hotels(df, city=None, min_safety=4.0, nights=1):
    """
    Filter hotels by city and minimum safety rating, compute total price for given nights,
    and sort by total price (cheapest first). Children are free (per specification),
    and price_per_night is used as the room's total per night.
    Returns a DataFrame with an added 'total_price' column.
    """
    # Defensive copy
    working = df.copy()

    # Filter by city if provided
    if city:
        working = working[working["city"].str.lower() == city.lower()]

    # Filter by safety rating threshold
    working = working[working["safety_rating"] >= float(min_safety)]

    # If nights invalid, correct to 1
    try:
        nights_int = int(nights)
        if nights_int < 1:
            nights_int = 1
    except Exception:
        nights_int = 1

    # Compute total price for the stay
    working = working.assign(total_price=(working["price_per_night"] * nights_int))

    # Sort cheapest first
    working = working.sort_values(by="total_price", ascending=True).reset_index(drop=True)
    return working

def render_results(df, nights, default_checkin_str, num_adults, num_children):
    """
    Render the filtered hotels as cards/rows. Each hotel row will display details and a 'Select' button.
    The cheapest result will be visually highlighted.
    Returns nothing, but uses Streamlit components and session state to handle selection.
    """
    st.markdown("### Available Hotels")
    if df.empty:
        st.warning("No hotels found matching your safety rating. Try lowering the safety filter or choosing another city.")
        return

    # Determine index of the cheapest option (first row after sorting)
    cheapest_idx = 0

    # Loop and render
    for idx, row in df.iterrows():
        # Unique button key
        select_key = f"select_hotel_{idx}"

        # Highlight cheapest option with a badge and different background
        is_cheapest = (idx == cheapest_idx)

        # Build a simple HTML card for nicer visuals
        card_html = f"""
        <div style="
            border:1px solid #ddd;
            border-radius:8px;
            padding:12px;
            margin-bottom:10px;
            background-color: {'#e6ffed' if is_cheapest else '#ffffff'};
        ">
            <div style="display:flex; justify-content:space-between; align-items:center;">
                <div style="flex:1;">
                    <h4 style="margin:0;">{row['hotel_name']} {'<span style=\"background:#ffeb3b;color:#000;padding:4px 8px;border-radius:6px;font-size:12px;\">Best Cheapest Option</span>' if is_cheapest else ''}</h4>
                    <div style="color:#444;margin-top:4px;">
                        <strong>City:</strong> {row['city']} &nbsp; • &nbsp;
                        <strong>Price/Night:</strong> ${row['price_per_night']} &nbsp; • &nbsp;
                        <strong>Safety:</strong> {row['safety_rating']} / 5.0
                    </div>
                    <div style="color:#666;margin-top:6px;">
                        {row['area_description']} &nbsp; • &nbsp; {row['distance_from_center_km']} km from center
                    </div>
                    <div style="margin-top:8px;"><strong>Total for {nights} night(s):</strong> ${row['total_price']}</div>
                </div>
            </div>
        </div>
        """

        st.markdown(card_html, unsafe_allow_html=True)

        # Place the select button immediately after the card
        col1, col2 = st.columns([1, 6])
        with col1:
            if st.button("Select", key=select_key):
                # Store selected hotel details in session state
                st.session_state["selected_hotel"] = {
                    "hotel_name": row["hotel_name"],
                    "city": row["city"],
                    "total_price": row["total_price"],
                    "price_per_night": row["price_per_night"],
                    "nights": nights,
                    "checkin_str": default_checkin_str,
                    "guests_str": f"{num_adults} Adults, {num_children} Children",
                }
                # After selecting, we show confirmation below (will be rendered by main loop)

        # spacing
        st.write("")

    # If a hotel was previously selected, render its confirmation
    if "selected_hotel" in st.session_state:
        sel = st.session_state["selected_hotel"]
        st.markdown("---")
        st.markdown("## Selected Hotel Confirmation")
        st.write(f"**Selected hotel name:** {sel['hotel_name']}")
        st.write(f"**City:** {sel['city']}")
        # The user requested the date display specifically as "30 December 2025"
        st.write(f"**Date:** {sel['checkin_str']}")
        st.write(f"**Guests:** {sel['guests_str']}")
        st.write(f"**Total amount:** ${sel['total_price']}")
        st.success("Your hotel selection has been recorded. (Demo only – no real payment).")

# Page config and title
st.set_page_config(page_title="Texas Hotel Booking – Family Trip (HYD → Texas)", layout="centered")
st.title("Texas Hotel Booking – Family Trip (HYD → Texas)")

# Top description (brief description of the trip)
st.markdown(
    """
    **Trip summary:** Family trip from Hyderabad (HYD) to Texas, USA.  
    Browse family-friendly hotels in Texas (Dallas, Houston, Austin, San Antonio), 
    filter by minimum safety rating, and choose the cheapest suitable option.
    """
)

# Main booking form (left column for inputs, right column for actions)
left_col, right_col = st.columns([2, 1])

with left_col:
    # Departure city (pre-filled, read-only)
    st.text_input("Departure city", value="Hyderabad (HYD)", disabled=True)

    # Destination state (pre-filled)
    st.text_input("Destination state", value="Texas, USA", disabled=True)

    # Check-in date (pre-filled as 2025-12-30 with a date picker)
    default_checkin = date(2025, 12, 30)
    checkin_date = st.date_input("Check-in date", value=default_checkin)

    # Number of nights (not explicitly required in prompt but necessary to compute total price)
    nights = st.number_input("Number of nights", min_value=1, max_value=30, value=3, step=1,
                             help="How many nights will you stay? (children stay free in this demo)")

    # Number of adults and children
    num_adults = st.number_input("Number of adults", min_value=1, max_value=10, value=2, step=1)
    num_children = st.number_input("Number of children", min_value=0, max_value=10, value=2, step=1)

    # City selectbox (Dallas, Houston, Austin, San Antonio)
    city_choice = st.selectbox("Choose city in Texas", options=["Dallas", "Houston", "Austin", "San Antonio"])

    # Safety rating filter (slider 1.0 to 5.0, default 4.0)
    min_safety = st.slider("Minimum safety rating", min_value=1.0, max_value=5.0, value=4.0, step=0.1)

with right_col:
    # Search button (user clicks to run filtering)
    st.write("")  # spacing
    st.write("")  # spacing
    # The user requested a button; label it "Search Hotels"
    search_clicked = st.button("Search Hotels", type="primary")

# Load data
hotel_df = load_hotel_data()

# Initialize session state for selection persistence
if "selected_hotel" not in st.session_state:
    st.session_state["selected_hotel"] = None

# When search clicked -> filter and show results
# Also allow auto-run if page reloaded and we have previous search state (but primary action via button)
if search_clicked:
    # Defensive validation for nights
    try:
        nights_int = int(nights)
        if nights_int < 1:
            st.warning("Number of nights must be at least 1. Using 1 night.")
            nights_int = 1
    except Exception:
        st.warning("Invalid number of nights. Using 1 night.")
        nights_int = 1

    # Filter and sort hotels
    filtered = filter_and_sort_hotels(hotel_df, city=city_choice, min_safety=min_safety, nights=nights_int)

    # Show brief trip description (as part of results area)
    st.markdown("### Trip details")
    st.write(f"**From:** Hyderabad (HYD)")
    st.write(f"**To:** Texas, USA ({city_choice})")
    # The confirmation is requested to display Date: 30 December 2025 for selected hotel; for informational purposes show chosen checkin too
    st.write(f"**Check-in chosen:** {checkin_date.strftime('%d %B %Y')}")
    st.write(f"**Guests:** {int(num_adults)} Adults, {int(num_children)} Children")
    st.write(f"**Minimum safety rating selected:** {min_safety}")

    # Render the filtered results and the selection UI
    render_results(filtered, nights_int, "30 December 2025", int(num_adults), int(num_children))
else:
    # No search yet: show a helpful hint and sample (do not automatically show hotel list)
    st.info("Enter your preferences on the left and click **Search Hotels** to find available options.")
    # Optionally show a small preview of available cities/prices (first 3 rows)
    st.markdown("#### Sample hotels (preview)")
    st.dataframe(hotel_df.head(5).loc[:, ["hotel_name", "city", "price_per_night", "safety_rating"]])
